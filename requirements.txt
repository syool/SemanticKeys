# Dataset loading (online only; offline uses the staged JSON).
#
# Pinned below 4.0 deliberately. datasets 4.x imports torchcodec at import
# time for audio/video decoding, and torchcodec needs FFmpeg shared libraries
# (libavutil et al.) present on the system. We do text only, so that is a
# dependency we gain nothing from — and on a container without FFmpeg it
# fails with "libavutil.so.56: cannot open shared object file", taking
# sentence-transformers down with it since it imports datasets.
#
# Staying on 3.x avoids needing apt packages, which matters for the offline
# bundle: wheels transfer cleanly, .deb files do not.
datasets>=2.20,<4.0
